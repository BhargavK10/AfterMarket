
# Employee Salary Sorting

n = int(input("Enter number of employees: "))

salary = []

for i in range(n):
    s = float(input("Enter salary: "))
    salary.append(s)

# 1. Selection Sort
a = salary.copy()

for i in range(n):
    min_index = i

    for j in range(i + 1, n):
        if a[j] < a[min_index]:
            min_index = j

    a[i], a[min_index] = a[min_index], a[i]

print("\nSelection Sort:", a)

print("Top 5 highest salaries:")
for i in range(n - 1, max(n - 6, -1), -1):
    print(a[i])


# 2. Bubble Sort
b = salary.copy()

for i in range(n):
    for j in range(0, n - i - 1):
        if b[j] > b[j + 1]:
            b[j], b[j + 1] = b[j + 1], b[j]

print("\nBubble Sort:", b)

print("Top 5 highest salaries:")
for i in range(n - 1, max(n - 6, -1), -1):
    print(b[i])
