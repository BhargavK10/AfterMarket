
# Hash Table Using Division Method and Chaining

SIZE = 10

hash_table = [[] for i in range(SIZE)]

# Hash Function
def hash_function(key):
    return key % SIZE

# Insert Operation
def insert(key, value):
    index = hash_function(key)

    for i in range(len(hash_table[index])):
        if hash_table[index][i][0] == key:
            hash_table[index][i] = (key, value)
            print("Value updated!")
            return

    hash_table[index].append((key, value))
    print("Key-value pair inserted!")

# Search Operation
def search(key):
    index = hash_function(key)

    for k, v in hash_table[index]:
        if k == key:
            print("Value found:", v)
            return

    print("Key not found!")

# Delete Operation
def delete(key):
    index = hash_function(key)

    for i in range(len(hash_table[index])):
        if hash_table[index][i][0] == key:
            hash_table[index].pop(i)
            print("Key deleted!")
            return

    print("Key not found!")

# Display Hash Table
def display():
    print("\nHash Table:")
    for i in range(SIZE):
        print(i, ":", hash_table[i])


# Main Program
while True:
    print("\n--- Hash Table Menu ---")
    print("1. Insert")
    print("2. Search")
    print("3. Delete")
    print("4. Display")
    print("5. Exit")

    choice = int(input("Enter choice: "))

    if choice == 1:
        key = int(input("Enter key: "))
        value = input("Enter value: ")
        insert(key, value)

    elif choice == 2:
        key = int(input("Enter key to search: "))
        search(key)

    elif choice == 3:
        key = int(input("Enter key to delete: "))
        delete(key)

    elif choice == 4:
        display()

    elif choice == 5:
        print("Exiting...")
        break

    else:
        print("Invalid choice!")
